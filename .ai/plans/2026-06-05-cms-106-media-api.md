# CMS-106 Media API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the first media API phase: project media settings, file upload to S3-compatible storage, media metadata persistence, metadata read, deletion, auth, and deterministic errors.

**Architecture:** Media assets are project-scoped records with environment-aware request routing for authorization consistency. Shared response/input contracts live in `@mdcms/shared`; the server owns media stores, object storage, route mounting, RBAC, and webhook emission.

**Tech Stack:** Bun, TypeScript, Elysia-style route mounting, Drizzle ORM, PostgreSQL, Zod 4, S3-compatible object storage through `fetch` and AWS Signature V4.

```ts
import { z } from "zod";

import { RuntimeError } from "../runtime/error.js";

export type MediaSettings = {
  media: {
    image: {
      maxUploadSizeBytes: number | null;
    };
  };
};

export type MediaAsset = {
  id: string;
  project: string;
  filename: string;
  mimeType: string;
  sizeBytes: number;
  url: string;
  uploadedBy: string;
  uploadedAt: string;
};

export type MediaSettingsResponse = { data: MediaSettings };
export type MediaAssetResponse = { data: MediaAsset };
export type MediaDeleteResponse = { data: { deleted: true; id: string } };

const PositiveSafeIntegerSchema = z
  .number()
  .int()
  .positive()
  .max(Number.MAX_SAFE_INTEGER);

const MediaSettingsSchema = z
  .object({
    media: z
      .object({
        image: z
          .object({
            maxUploadSizeBytes: PositiveSafeIntegerSchema.nullable(),
          })
          .strict(),
      })
      .strict(),
  })
  .strict();

const NonEmptyStringSchema = z.string().trim().min(1);
const MediaAssetSchema = z
  .object({
    id: NonEmptyStringSchema,
    project: NonEmptyStringSchema,
    filename: NonEmptyStringSchema,
    mimeType: NonEmptyStringSchema,
    sizeBytes: PositiveSafeIntegerSchema.or(z.literal(0)),
    url: NonEmptyStringSchema,
    uploadedBy: NonEmptyStringSchema,
    uploadedAt: NonEmptyStringSchema,
  })
  .strict();

const MediaSettingsResponseSchema = z
  .object({ data: MediaSettingsSchema })
  .strict();
const MediaAssetResponseSchema = z.object({ data: MediaAssetSchema }).strict();
const MediaDeleteResponseSchema = z
  .object({
    data: z
      .object({
        deleted: z.literal(true),
        id: NonEmptyStringSchema,
      })
      .strict(),
  })
  .strict();

function runtimeError(message: string, details?: Record<string, unknown>) {
  return new RuntimeError({
    code: "INVALID_INPUT",
    message,
    statusCode: 400,
    ...(details ? { details } : {}),
  });
}

function assertRecord(value: unknown): Record<string, unknown> {
  if (typeof value !== "object" || value === null || Array.isArray(value)) {
    throw runtimeError("Media settings payload must be an object.", {
      field: "body",
    });
  }
  return value as Record<string, unknown>;
}

function rejectUnknownFields(
  record: Record<string, unknown>,
  allowed: readonly string[],
): void {
  const allowedSet = new Set(allowed);
  const unknown = Object.keys(record).find((field) => !allowedSet.has(field));
  if (unknown) {
    throw runtimeError(`Field "${unknown}" is not allowed.`, {
      field: unknown,
    });
  }
}

export function createDefaultMediaSettings(): MediaSettings {
  return { media: { image: { maxUploadSizeBytes: null } } };
}

export function parseMediaSettingsInput(value: unknown): MediaSettings {
  const root = assertRecord(value);
  rejectUnknownFields(root, ["media"]);
  const parsed = MediaSettingsSchema.safeParse(root);
  if (!parsed.success) {
    throw runtimeError(
      "media.image.maxUploadSizeBytes must be null or a positive safe integer in bytes.",
      { field: "media.image.maxUploadSizeBytes" },
    );
  }
  return parsed.data;
}

export function assertMediaSettingsResponse(
  value: unknown,
): asserts value is MediaSettingsResponse {
  if (!MediaSettingsResponseSchema.safeParse(value).success) {
    throw runtimeError("Value must be a media settings response.");
  }
}

export function assertMediaAssetResponse(
  value: unknown,
): asserts value is MediaAssetResponse {
  if (!MediaAssetResponseSchema.safeParse(value).success) {
    throw runtimeError("Value must be a media asset response.");
  }
}

export function assertMediaDeleteResponse(
  value: unknown,
): asserts value is MediaDeleteResponse {
  if (!MediaDeleteResponseSchema.safeParse(value).success) {
    throw runtimeError("Value must be a media delete response.");
  }
}
```

```ts
export function parseMediaId(value: unknown): string {
```
